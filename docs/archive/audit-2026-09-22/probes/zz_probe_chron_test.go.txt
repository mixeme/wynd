package chronicle_test

import (
	"errors"
	"testing"
	"time"

	"gitea.mixdep.ru/mix/wynd/internal/chronicle"
)

// Probe 04-1: foreign client_id returns another author's post.
func TestProbeClientIDForeignAuthor(t *testing.T) {
	e := newTestEnv(t)
	circle := e.createCircle("owner", "Аня", chronicle.DurationWindow(30*24*time.Hour))
	e.join(circle.ID, "bob", "Боб", e.after(time.Hour))
	a, err := e.ch.CreatePost(e.ctx, chronicle.PostInput{
		CircleID: circle.ID, AccountID: "owner", Body: "секрет", EntryDate: "2026-08-01", Now: e.after(2 * time.Hour), ClientID: "k1",
	})
	if err != nil {
		t.Fatal(err)
	}
	b, err := e.ch.CreatePost(e.ctx, chronicle.PostInput{
		CircleID: circle.ID, AccountID: "bob", Body: "x", EntryDate: "2026-08-01", Now: e.after(3 * time.Hour), ClientID: "k1",
	})
	t.Logf("bob got err=%v id=%s body=%q (owner's id=%s)", err, b.ID, b.Body, a.ID)
	if err == nil && b.ID == a.ID {
		t.Fatalf("CONFIRMED: bob received owner's post via client_id")
	}
}

// Probe 04-2: excluded member clears day title.
func TestProbeExcludedClearsDayTitle(t *testing.T) {
	e := newTestEnv(t)
	circle := e.createCircle("owner", "Аня", chronicle.DurationWindow(30*24*time.Hour))
	e.join(circle.ID, "bob", "Боб", e.after(time.Hour))
	e.post(circle.ID, "bob", "моё", "2026-08-01", e.after(2*time.Hour))
	e.post(circle.ID, "owner", "и моё", "2026-08-01", e.after(2*time.Hour))
	if err := e.ch.SetDayTitle(e.ctx, chronicle.DayTitleInput{CircleID: circle.ID, AccountID: "owner", EntryDate: "2026-08-01", Title: "Заголовок", Now: e.after(3 * time.Hour)}); err != nil {
		t.Fatal(err)
	}
	if err := e.ch.Exclude(e.ctx, circle.ID, "owner", "bob", e.after(4*time.Hour)); err != nil {
		t.Fatal(err)
	}
	err := e.ch.ClearDayTitle(e.ctx, circle.ID, "bob", "2026-08-01", e.after(5*time.Hour))
	t.Logf("ClearDayTitle by excluded: err=%v", err)
	if !errors.Is(err, chronicle.ErrForbidden) {
		var title string
		_ = e.ch.DB().QueryRowContext(e.ctx, `SELECT COALESCE(title,'') FROM days WHERE circle_id = ? AND entry_date = ?`, circle.ID, "2026-08-01").Scan(&title)
		t.Fatalf("CONFIRMED: excluded cleared day title; err=%v title now=%q", err, title)
	}
}
