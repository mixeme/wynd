package search_test

import (
	"testing"
	"time"

	"gitea.mixdep.ru/mix/wynd/internal/chronicle"
	"gitea.mixdep.ru/mix/wynd/internal/search"
	"gitea.mixdep.ru/mix/wynd/internal/store"
)

func TestReviewSearch(t *testing.T) {
	st, _ := store.OpenMemory()
	defer st.Close()
	ch, _ := chronicle.New(st)
	svc := search.New(ch)
	ctx := t.Context()
	now := time.Date(2026, 8, 30, 10, 0, 0, 0, time.UTC)
	circle, _, _, err := ch.CreateCircle(ctx, chronicle.CreateCircleInput{Name: "Семья", OwnerAccountID: "owner", OwnerName: "Аня", Now: now})
	if err != nil {
		t.Fatal(err)
	}
	post, err := ch.CreatePost(ctx, chronicle.PostInput{CircleID: circle.ID, AccountID: "owner", Body: "старый пост про море и лето на даче", EntryDate: "2026-08-29", Now: now})
	if err != nil {
		t.Fatal(err)
	}
	if _, _, err = ch.Join(ctx, chronicle.JoinInput{CircleID: circle.ID, AccountID: "bob", Name: "Боб", Now: now.Add(time.Hour)}); err != nil {
		t.Fatal(err)
	}
	if _, err := ch.CreateComment(ctx, chronicle.CommentInput{CircleID: circle.ID, AccountID: "owner", PostID: post.ID, Body: "секретный комментарий", Now: now.Add(2 * time.Hour)}); err != nil {
		t.Fatal(err)
	}
	feed, err := ch.FeedSnapshot(ctx, circle.ID, "bob")
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("bob feed posts: %d", len(feed))
	hits, err := svc.SearchCircle(ctx, "bob", circle.ID, "секретный", 10, search.Filters{})
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("bob hits for comment on invisible post: %d %+v", len(hits), hits)

	// multi-word non-adjacent
	hits, err = svc.SearchCircle(ctx, "owner", circle.ID, "море дача", 10, search.Filters{})
	t.Logf("non-adjacent 'море дача' hits=%d err=%v", len(hits), err)
	hits, err = svc.SearchCircle(ctx, "owner", circle.ID, "море даче", 10, search.Filters{})
	t.Logf("non-adjacent 'море даче' hits=%d err=%v", len(hits), err)
	for _, q := range []string{`"`, `*`, `!!!`, `a"b`, `NEAR(`, `-`, `море*`, "мор", "a\x00b", `AND`, `(`} {
		hits, err = svc.SearchCircle(ctx, "owner", circle.ID, q, 10, search.Filters{})
		t.Logf("q=%q hits=%d err=%v", q, len(hits), err)
	}
}
