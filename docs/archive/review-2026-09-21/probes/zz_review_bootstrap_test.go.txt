package api_test

import (
	"bytes"
	"encoding/json"
	"net/http"
	"net/http/httptest"
	"os"
	"path/filepath"
	"testing"
)

// Review probe: does POST /admin/bootstrap persist SMTP + public_url before the token is checked?
func TestReviewBootstrapSideEffectsBeforeToken(t *testing.T) {
	srv, _, _, _ := setupAPI(t) // already bootstrapped
	body, _ := json.Marshal(map[string]any{
		"token":         "WRONG",
		"password":      "whatever-pass",
		"public_url":    "https://evil.example",
		"host":          "127.0.0.1",
		"port":          1,
		"username":      "u",
		"smtp_password": "p",
		"from":          "attacker@evil.example",
	})
	req := httptest.NewRequest(http.MethodPost, "/api/v1/admin/bootstrap", bytes.NewReader(body))
	rec := httptest.NewRecorder()
	srv.ServeHTTP(rec, req)
	t.Logf("status=%d body=%s", rec.Code, rec.Body.String())

	cfg, err := srv.Mail.LoadConfig(t.Context())
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("smtp host after=%q from=%q", cfg.Host, cfg.From)
	data, _ := os.ReadFile(filepath.Join(srv.DataDir, "config.json"))
	t.Logf("config.json after=%s", string(data))
	t.Logf("srv.PublicURL after=%q loopback=%v", srv.PublicURL, srv.Loopback)
	if cfg.Host != "" || srv.PublicURL != "http://127.0.0.1:7676" {
		t.Fatalf("CONFIRMED: unauthenticated bootstrap call mutated state")
	}
}
