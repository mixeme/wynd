func TestReviewAvatarSurvivesRoutine(t *testing.T) {
	st, _ := store.OpenMemory()
	defer st.Close()
	ch, _ := chronicle.New(st)
	dir := t.TempDir()
	s, _ := blob.New(st, filepath.Join(dir, "blobs"))
	_ = os.MkdirAll(s.Dir(), 0o750)
	ctx := t.Context()
	seedAccount(t, st, "owner")
	t0 := time.Now().UTC().Add(-72 * time.Hour)
	circle, _, _, err := ch.CreateCircle(ctx, chronicle.CreateCircleInput{Name: "Семья", OwnerAccountID: "owner", OwnerName: "Аня", Now: t0})
	if err != nil {
		t.Fatal(err)
	}
	face := uploadComplete(t, s, "owner", "image/jpeg", []byte("owner-face"))
	if _, err := s.DB().Exec(`UPDATE blobs SET created_at = ? WHERE id = ?`, t0.Format(time.RFC3339Nano), face.ID); err != nil {
		t.Fatal(err)
	}
	if err := ch.UpdateIdentity(ctx, circle.ID, "owner", chronicle.UpdateIdentityInput{AvatarBlobID: &face.ID}, t0); err != nil {
		t.Fatal(err)
	}
	counts, err := jobs.RunDailyRoutine(ctx, s.DB(), s.Dir(), time.Now().UTC())
	t.Logf("counts=%+v err=%v", counts, err)
	if _, err := s.OpenBlob(ctx, face.ID); err != nil {
		t.Fatalf("AVATAR BLOB GONE after routine: %v", err)
	}
}

func TestReviewPurgeFreesBlob(t *testing.T) {
	st, _ := store.OpenMemory()
	defer st.Close()
	ch, _ := chronicle.New(st)
	dir := t.TempDir()
	s, _ := blob.New(st, filepath.Join(dir, "blobs"))
	_ = os.MkdirAll(s.Dir(), 0o750)
	ctx := t.Context()
	seedAccount(t, st, "owner")
	t0 := time.Date(2026, 8, 1, 10, 0, 0, 0, time.UTC)
	circle, _, _, err := ch.CreateCircle(ctx, chronicle.CreateCircleInput{Name: "Семья", OwnerAccountID: "owner", OwnerName: "Аня", Now: t0})
	if err != nil {
		t.Fatal(err)
	}
	b := uploadComplete(t, s, "owner", "image/jpeg", []byte("old-photo"))
	post, err := ch.CreatePost(ctx, chronicle.PostInput{CircleID: circle.ID, AccountID: "owner", Body: "фото", EntryDate: "2026-08-01", Now: t0})
	if err != nil {
		t.Fatal(err)
	}
	if err := ch.AttachMedia(ctx, post.ID, []chronicle.MediaInput{{BlobID: b.ID, Kind: chronicle.MediaPhoto}}); err != nil {
		t.Fatal(err)
	}
	if err := s.AddRef(ctx, nil, b.ID, "post", post.ID); err != nil {
		t.Fatal(err)
	}
	deadline := t0.Add(48 * time.Hour)
	if err := ch.StartArchiveCycle(ctx, circle.ID, "owner", "2026-08-02", deadline, 0, t0.Add(time.Hour)); err != nil {
		t.Fatal(err)
	}
	counts, err := jobs.RunArchiveJobs(ctx, s.DB(), ch, s, nil, "http://x", deadline.Add(time.Second))
	if err != nil || counts.PurgedCircles != 1 {
		t.Fatalf("purge: %+v %v", counts, err)
	}
	if _, err := s.OpenBlob(ctx, b.ID); err != blob.ErrNotFound {
		t.Fatalf("BLOB NOT FREED after archive purge: err=%v", err)
	}
}

func TestReviewRoutineAfterPayScreenshotCleanup(t *testing.T) {
	st, _ := store.OpenMemory()
	defer st.Close()
	dir := t.TempDir()
	s, _ := blob.New(st, filepath.Join(dir, "blobs"))
	_ = os.MkdirAll(s.Dir(), 0o750)
	ctx := t.Context()
	seedAccount(t, st, "payer")
	old := time.Now().UTC().Add(-72 * time.Hour).Format(time.RFC3339Nano)
	if _, err := s.DB().Exec(`INSERT INTO blobs (id, account_id, sha256, size_bytes, mime_type, storage_path, status, created_at)
		VALUES ('blob-pay', 'payer', 'abc', 4, 'image/jpeg', 'pa/shot', 'complete', ?)`, old); err != nil {
		t.Fatal(err)
	}
	// state after CleanupExpiredPayScreenshots: blob_deleted=1, no blob_refs, row kept
	if _, err := s.DB().Exec(`INSERT INTO pay_requests (id, account_id, blob_id, status, created_at, resolved_at, blob_deleted)
		VALUES ('req-1', 'payer', 'blob-pay', 'approved', ?, ?, 1)`, old, old); err != nil {
		t.Fatal(err)
	}
	_, err := jobs.RunDailyRoutine(ctx, s.DB(), s.Dir(), time.Now().UTC())
	if err != nil {
		t.Fatalf("ROUTINE FAILS after pay screenshot cleanup: %v", err)
	}
}
