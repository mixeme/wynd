package chronicle_test

import (
	"testing"
	"time"

	"gitea.mixdep.ru/mix/wynd/internal/chronicle"

func TestReviewExcludeLeftWithAccess(t *testing.T) {
	ch := newTestChronicle(t)
	ctx := t.Context()
	now := time.Date(2026, 8, 30, 10, 0, 0, 0, time.UTC)
	circle, _, _, err := ch.CreateCircle(ctx, chronicle.CreateCircleInput{Name: "Семья", OwnerAccountID: "owner", OwnerName: "Аня", Now: now})
	if err != nil {
		t.Fatal(err)
	}
	if _, _, err := ch.Join(ctx, chronicle.JoinInput{CircleID: circle.ID, AccountID: "bob", Name: "Боб", Now: now.Add(time.Hour)}); err != nil {
		t.Fatal(err)
	}
	if err := ch.LeaveWithAccess(ctx, circle.ID, "bob", now.Add(2*time.Hour)); err != nil {
		t.Fatal(err)
	}
	err = ch.Exclude(ctx, circle.ID, "owner", "bob", now.Add(3*time.Hour))
	t.Logf("Exclude(left_with_access) err=%v", err)
	canRead, err2 := ch.CanReadEvent(ctx, circle.ID, "bob", now.Add(90*time.Minute))
	t.Logf("bob can still read his span: %v (%v)", canRead, err2)
	if err != nil {
		t.Fatalf("OWNER CANNOT REVOKE ACCESS of left_with_access member: %v", err)
	}
}

func TestReviewExcludedCanStillEditAndDelete(t *testing.T) {
	ch := newTestChronicle(t)
	ctx := t.Context()
	now := time.Date(2026, 8, 30, 10, 0, 0, 0, time.UTC)
	circle, _, _, err := ch.CreateCircle(ctx, chronicle.CreateCircleInput{Name: "Семья", OwnerAccountID: "owner", OwnerName: "Аня", Now: now})
	if err != nil {
		t.Fatal(err)
	}
	if _, _, err := ch.Join(ctx, chronicle.JoinInput{CircleID: circle.ID, AccountID: "bob", Name: "Боб", Now: now.Add(time.Hour)}); err != nil {
		t.Fatal(err)
	}
	post, err := ch.CreatePost(ctx, chronicle.PostInput{CircleID: circle.ID, AccountID: "bob", Body: "моё", EntryDate: "2026-08-30", Now: now.Add(2 * time.Hour)})
	if err != nil {
		t.Fatal(err)
	}
	if _, err := ch.CreateComment(ctx, chronicle.CommentInput{CircleID: circle.ID, AccountID: "owner", PostID: post.ID, Body: "ответ владельца", Now: now.Add(3 * time.Hour)}); err != nil {
		t.Fatal(err)
	}
	if err := ch.Exclude(ctx, circle.ID, "owner", "bob", now.Add(4*time.Hour)); err != nil {
		t.Fatal(err)
	}
	errEdit := ch.EditPost(ctx, circle.ID, "bob", post.ID, "переписал после исключения", "", now.Add(5*time.Hour))
	t.Logf("EditPost by excluded: err=%v", errEdit)
	errDel := ch.DeletePost(ctx, circle.ID, "bob", post.ID, now.Add(6*time.Hour))
	t.Logf("DeletePost by excluded: err=%v", errDel)
	if errEdit == nil || errDel == nil {
		t.Fatalf("EXCLUDED MEMBER STILL WRITES: edit=%v delete=%v", errEdit, errDel)
	}
}

func TestReviewDeleteCircle(t *testing.T) {
	ch := newTestChronicle(t)
	ctx := t.Context()
	now := time.Date(2026, 8, 30, 10, 0, 0, 0, time.UTC)
	circle, _, _, err := ch.CreateCircle(ctx, chronicle.CreateCircleInput{Name: "Семья", OwnerAccountID: "owner", OwnerName: "Аня", Now: now})
	if err != nil {
		t.Fatal(err)
	}
	post, err := ch.CreatePost(ctx, chronicle.PostInput{CircleID: circle.ID, AccountID: "owner", Body: "уникальноеслово", EntryDate: "2026-08-30", Now: now})
	if err != nil {
		t.Fatal(err)
	}
	if _, err := ch.CreateComment(ctx, chronicle.CommentInput{CircleID: circle.ID, AccountID: "owner", PostID: post.ID, Body: "комментслово", Now: now.Add(time.Minute)}); err != nil {
		t.Fatal(err)
	}
	if _, err := ch.SetReaction(ctx, chronicle.ReactionInput{CircleID: circle.ID, AccountID: "owner", PostID: post.ID, Emoji: "heart", Now: now.Add(2 * time.Minute)}); err != nil {
		t.Fatal(err)
	}
	if err := ch.SetDayTitle(ctx, chronicle.DayTitleInput{CircleID: circle.ID, AccountID: "owner", EntryDate: "2026-08-30", Title: "деньслово", Now: now.Add(3 * time.Minute)}); err != nil {
		t.Fatal(err)
	}
	var ftsBefore int
	_ = ch.DB().QueryRow(`SELECT count(*) FROM content_fts WHERE circle_id = ?`, circle.ID).Scan(&ftsBefore)
	err = ch.DeleteCircle(ctx, circle.ID, "owner", "Семья", now.Add(time.Hour))
	t.Logf("DeleteCircle err=%v", err)
	var ftsAfter, posts, events int
	_ = ch.DB().QueryRow(`SELECT count(*) FROM content_fts WHERE circle_id = ?`, circle.ID).Scan(&ftsAfter)
	_ = ch.DB().QueryRow(`SELECT count(*) FROM posts WHERE circle_id = ?`, circle.ID).Scan(&posts)
	_ = ch.DB().QueryRow(`SELECT count(*) FROM events WHERE circle_id = ?`, circle.ID).Scan(&events)
	t.Logf("fts before=%d after=%d posts=%d events=%d", ftsBefore, ftsAfter, posts, events)
	if err != nil || ftsAfter != 0 {
		t.Fatalf("DELETE CIRCLE PROBLEM: err=%v ftsAfter=%d", err, ftsAfter)
	}
}
